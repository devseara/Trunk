function	script	F_BPEXP	{
	.@exp_value = getarg(0);
	.@max_exp = $bpExpPerLevel * $bpMaxLevel;
	.@old_level = #BPEXP / $bpExpPerLevel;

	if (#BPEXP < .@max_exp) {
		set #BPEXP, #BPEXP + .@exp_value;
		.@new_level = #BPEXP / $bpExpPerLevel;
		.@exp_in_level = #BPEXP % $bpExpPerLevel;
		dispbottom "[ Battle Pass ] : +" + callfunc("F_InsertComma", .@exp_value) + " EXP (" + callfunc("F_InsertComma", .@exp_in_level) + " / " + callfunc("F_InsertComma", $bpExpPerLevel) + ")";
		if (.@new_level > .@old_level) {
			announce "[ Battle Pass ] : " + strcharinfo(0) + " has leveled up their Battle Pass to Level " + (.@new_level + 1) + "!", bc_all, 0x00FF00;
		}
	}
	return;
}

function	script	F_ItemDisp2	{
    .@item_id = getarg(0);
    .@item_name$ = "^i[" + .@item_id + "]" + getitemname(.@item_id);
    .@slots = getitemslots(.@item_id);
    if (.@slots > 0) {
        return .@item_name$ + " [" + .@slots + "]";
    } else {
        return .@item_name$;
    }
}

///////////////////////////////////////////////////////////////////////////////////////
// - Additional Function for NPC Automatic Add the Useful variable.
// - F_npcvaradd("#TESTVARIABLE",1);
// - 1 - Daily | 2 - Weekly | 3 - Monthly
///////////////////////////////////////////////////////////////////////////////////////
function	script	F_npcvaradd	{
    .@new_key$ = getarg(0);
    .@varmode = getarg(1);

    if (.@varmode == 3) {
        .@size = getarraysize($monthly_reset_keys$);
        for (.@i = 0; .@i < .@size; .@i++) {
            if ($monthly_reset_keys$[.@i] == .@new_key$)
                return;
        }
        setarray $monthly_reset_keys$[.@size], .@new_key$;
    } else if (.@varmode == 2) {
        .@size = getarraysize($weekly_reset_keys$);
        for (.@i = 0; .@i < .@size; .@i++) {
            if ($weekly_reset_keys$[.@i] == .@new_key$)
                return;
        }
        setarray $weekly_reset_keys$[.@size], .@new_key$;
    } else { // Daily
        .@size = getarraysize($daily_reset_keys$);
        for (.@i = 0; .@i < .@size; .@i++) {
            if ($daily_reset_keys$[.@i] == .@new_key$)
                return;
        }
        setarray $daily_reset_keys$[.@size], .@new_key$;
    }
    return;
}

        mes "<b>^FF9933You are Premium^000000.</b>";

    set .@tsegments, 70;
    set .@segmentEXP, .@maxEXP / .@tsegments;
    set .@fsegments, min(.@tsegments, #BPEXP / .@segmentEXP);

    set .@progressBar$, "[ ";
    for (set .@i, 0; .@i < .@tsegments; set .@i, .@i + 1) {
        if (.@i < .@fsegments) {

                set .@progressBar$, .@progressBar$ + "^00FF00ll^000000"; // Green
            else
                set .@progressBar$, .@progressBar$ + "^FF9933ll^000000"; // Orange

            set .@progressBar$, .@progressBar$ + "^a0a0a0ll^000000"; // Gray
        }

    set .@decimalPart, .@percentage % 100;

    mes "Level Progress : " + .@levelBar$ + " (" + F_InsertComma(.@levelHaveExp) + " / " + F_InsertComma(.@levelNeedExp) + " ) EXP";
    mes " ";

    for (set .@i, 0; .@i < getarraysize(.rewards); set .@i, .@i + 3) {

        if (.@lvl > .@currentLevel) {

                set .@nextNormLevel, .@lvl;
        }

    if (.@nextNormLevel) {
        set .@msg$, "";
        for (set .@i, 0; .@i < getarraysize(.rewards); set .@i, .@i + 3) {
            if (.rewards[.@i] == .@nextNormLevel) {

                set .@qty,    .rewards[.@i + 2];
                if (.@msg$ != "")
                    set .@msg$, .@msg$ + " | ";
                set .@msg$, .@msg$ + callfunc("F_ItemDisp2", .@itemid) + " x" + .@qty;
            }
        }
        mes "Next Normal Reward at Level " + .@nextNormLevel + " : " + .@msg$;
    } else {
        mes "You have obtained all normal rewards.";
    }

    if (#BPPREMIUM) {
        set .@nextPremLevel, 0;

        for (set .@i, 0; .@i < getarraysize(.prewards); set .@i, .@i + 3) {
            set .@lvl, .prewards[.@i];
            if (.@lvl > .@currentLevel) {
                if (.@nextPremLevel == 0 || .@lvl < .@nextPremLevel)
                    set .@nextPremLevel, .@lvl;
            }
        }

        if (.@nextPremLevel) {

            for (set .@i, 0; .@i < getarraysize(.prewards); set .@i, .@i + 3) {

            mes "Next ^FF9933Premium^000000 Reward at Level " + .@nextPremLevel + " : " + .@pmsg$;
        } else {
            mes "You have obtained all premium rewards.";
        }

    .@select = select("View Rewards", "Claim Rewards","Battle Pass Quest", "Cancel");

                callfunc "F_NPC_NAME";

                set .@i, 0;

                while (.@i < getarraysize(.rewards)) {
                    set .@level, .rewards[.@i];
                    set .@plevel, .prewards[.@z];
                    if (.@level <= .@currentLevel)
                        mes "Level : " + .@level + " ( ^00FF00Completed^000000 )";
                    else
                        mes "Level : " + .@level;

                    set .@itemList$, "";
                    set .@levelItems, 0;
                    while (.@i + 1 < getarraysize(.rewards) && .rewards[.@i] == .@level) {
                        set .@itemid, .rewards[.@i + 1];
                        set .@quantity, .rewards[.@i + 2];
                        if (.@levelItems > 0) {
                            set .@itemList$, .@itemList$ + " | ";
                        }
                        set .@itemList$, .@itemList$ + callfunc("F_ItemDisp2", .@itemid) + " x" + .@quantity;
                        set .@i, .@i + 3;
                        set .@levelItems, .@levelItems + 1;
                    }

                    set .@pitemList$, "";
                    set .@plevelItems, 0;
                    while (.@z + 1 < getarraysize(.prewards) && .prewards[.@z] == .@plevel) {
                        set .@pitemid, .prewards[.@z + 1];
                        set .@pquantity, .prewards[.@z + 2];
                        if (.@plevelItems > 0) {
                            set .@pitemList$, .@pitemList$ + " | ";
                        }
                        set .@pitemList$, .@pitemList$ + callfunc("F_ItemDisp2", .@pitemid) + " x" + .@pquantity;
                        set .@z, .@z + 3;
                        set .@plevelItems, .@plevelItems + 1;
                    }

                    mes .@itemList$;
                    mes "^FF9933Premium Rewards^000000 ------------";
                    mes .@pitemList$;
                    mes "------------------------------------------------------------------------------------------";

                mes "Thats All!";
                if(select("^0055FFReturn^000000:Close") == 2) end;
                    clear;
                    goto OnInteract;
            break;
            case 2:
                setdialogsize(280, 182);
                setarray .@lmReward, 0;
                set .@rewardCount, 0;
                if (!#BPPREMIUM)
                    goto NonPremium;
                clear;

                    mes " ";

                    set .@selectedLevel, .@levels[.@selected];

                    for (set .@i, 0; .@i < .@size; set .@i, .@i + 3) {
