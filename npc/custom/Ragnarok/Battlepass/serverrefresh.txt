-	script	Server Refresh	FAKE_NPC,{

function F_ResetVar;
function F_KeyMenu;

OnInit:
    bindatcmd "dayrefresh", strnpcinfo(0)+"::OnMBReset1", 99,99;
    bindatcmd "weekrefresh", strnpcinfo(0)+"::OnMBReset2", 99,99;
    bindatcmd "monthrefresh", strnpcinfo(0)+"::OnMBReset3", 99,99;
    bindatcmd "allrefresh", strnpcinfo(0)+"::OnMBReset", 99,99;
    bindatcmd "srconf", strnpcinfo(0)+"::OnCommand", 99,99;
    if (!$servertimereset) $servertimereset = 6; // Default reset time is 6:AM
    if (!getarraysize($reset_keys$)) {
        $reset_keys$[0] = "";
    }
end;

OnMinute00:
    if (gettime(DT_HOUR) == $servertimereset)
        F_ResetVar("daily");
    if (gettime(DT_DAYOFWEEK) == MONDAY && gettime(DT_HOUR) == $servertimereset)
        F_ResetVar("weekly");
    if (gettime(DT_DAYOFMONTH) == 1 && gettime(DT_HOUR) == $servertimereset)
        F_ResetVar("monthly");
end;

OnMBReset3:
    F_ResetVar("monthly");
    announce "[ System ] : Monthlies refreshed by Administrator.",bc_all;
end;
OnMBReset2:
    F_ResetVar("weekly");
    announce "[ System ] : Weeklies refreshed by Administrator.",bc_all;
end;
OnMBReset1:
    F_ResetVar("daily");
    announce "[ System ] : Dailies refreshed by Administrator.",bc_all;
end;

OnMBReset:
    F_ResetVar("monthly");
    F_ResetVar("weekly");
    F_ResetVar("daily");
    announce "[ System ] : Server refreshed by Administrator.",bc_all;
end;

OnCommand:
    mes "[ " + strnpcinfo(1) + " ] : Server Setup Menu";
    mes " ";
    if ($servertimereset == 0) {
        mes " - ^FF0000Server will reset at 12 AM (midnight)^000000.";
    } else if ($servertimereset < 12) {
        mes " - ^FF0000Server will reset at " + $servertimereset + " AM^000000.";
    } else if ($servertimereset == 12) {
        mes " - ^FF0000Server will reset at 12 PM (noon)^000000.";
    } else {
        mes " - ^FF0000Server will reset at " + ($servertimereset - 12) + " PM^000000.";
    }
    switch(select("Set Server Refresh Time:Manage Reset Keys")) {
            case 1:
            clear;
            mes "[ " + strnpcinfo(1) + " ]";
            mes "Enter new reset time (1 to 24) : ";
            input .@new_reset_time, 1, 24;
            $servertimereset = .@new_reset_time;
            clear;
            mes "[ " + strnpcinfo(1) + " ]";
            if ($servertimereset == 0) {
                mes "Server reset time has been set to 12 AM (midnight).";
            } else if ($servertimereset < 12) {
                mes "Server reset time has been set to " + $servertimereset + " AM.";
            } else if ($servertimereset == 12) {
                mes "Server reset time has been set to 12 PM (noon).";
            } else {
                mes "Server reset time has been set to " + ($servertimereset - 12) + " PM.";
            }
            close;
        case 2:
            F_KeyMenu();
            close;
    }
    close;

    function	F_KeyMenu	{
        .@type$ = "";
        .@type2$ = "";
        while (true) {
            clear;
            mes "[Reset Key Manager]";
            mes "Choose the reset type:";
            .@choice = select("Daily Reset", "Weekly Reset", "Monthly Reset", "Cancel");
            if (.@choice == 4) end;

            switch (.@choice) {
                case 1: .@type$ = "$daily_reset_keys$"; .@type2$ = "Daily Reset"; break;
                case 2: .@type$ = "$weekly_reset_keys$"; .@type2$ = "Weekly Reset"; break;
                case 3: .@type$ = "$monthly_reset_keys$"; .@type2$ = "Monthly Reset"; break;
            }

            while (true) {
                // Load keys manually
                deletearray .@key_list$[0], getarraysize(.@key_list$);
                .@key_size = getarraysize(getd(.@type$));
                for (.@i = 0; .@i < .@key_size; .@i++) {
                    .@key_list$[.@i] = getd(.@type$ + "[" + .@i + "]");
                }

                clear;
                mes "[Key List]";
                mes "Reset Type: ^0000FF" + .@type2$ + "^000000";

                if (.@key_size == 0) {
                    mes "No keys set.";
                } else {
                    for (.@i = 0; .@i < .@key_size; .@i++) {
                        mes (.@i+1) + ". " + .@key_list$[.@i];
                    }
                }
                next;

                .@action = select("Add Key", "Edit Key", "Delete Key", "Back");

                if (.@action == 4) break;

                if (.@action == 1) {
                    // Add Key
                    clear;
                    mes "[Key List]";
                    mes "Reset Type: ^0000FF" + .@type2$ + "^000000";
                    if (.@key_size == 0) {
                        mes "No keys set.";
                    } else {
                        for (.@i = 0; .@i < .@key_size; .@i++) {
                            mes (.@i+1) + ". " + .@key_list$[.@i];
                        }
                    }
                    mes "^FF0000Enter new key to add:^000000";
                    input .@newkey$;
                    if (.@newkey$ == "") continue;

                    // Reload list for duplicate check
                    .@found = 0;
                    for (.@i = 0; .@i < .@key_size; .@i++)
                        if (.@key_list$[.@i] == .@newkey$) .@found = 1;

                    if (.@found) {
                        mes "Key already exists.";
                        next;
                        continue;
                    }

                    set getd(.@type$ + "[" + .@key_size + "]"), .@newkey$;
                    mes "Key added!";
                    next;

                } else if (.@action == 2) {
                    // Edit Key
                    if (.@key_size == 0) continue;
                    clear;
                    mes "[Key List]";
                    mes "Reset Type: ^0000FF" + .@type2$ + "^000000";
                    mes "^FF0000Choose key number to edit:^000000";
                    if (.@key_size == 0) {
                        mes "No keys set.";
                    } else {
                        for (.@i = 0; .@i < .@key_size; .@i++) {
                            mes (.@i+1) + ". " + .@key_list$[.@i];
                        }
                    }
                    input .@edit;
                    if (.@edit <= 0 || .@edit > .@key_size) continue;
                    mes "Enter new value for key:";
                    input .@newval$;
                    if (.@newval$ == "") continue;

                    // Check if already exists
                    .@found = 0;
                    for (.@i = 0; .@i < .@key_size; .@i++)
                        if (.@key_list$[.@i] == .@newval$) .@found = 1;

                    if (.@found) {
                        mes "Key already exists.";
                        next;
                        continue;
                    }

                    set getd(.@type$ + "[" + (.@edit - 1) + "]"), .@newval$;
                    mes "Key updated!";
                    next;

                } else if (.@action == 3) {
                    // Delete Key
                    if (.@key_size == 0) continue;
                    clear;
                    mes "[Key List]";
                    mes "Reset Type: ^0000FF" + .@type2$ + "^000000";
                    if (.@key_size == 0) {
                        mes "No keys set.";
                    } else {
                        for (.@i = 0; .@i < .@key_size; .@i++) {
                            mes (.@i+1) + ". " + .@key_list$[.@i];
                        }
                    }
                    mes "^FF0000Choose key number to delete:^000000";
                    input .@del;
                    if (.@del <= 0 || .@del > .@key_size) continue;
                    deletearray getd(.@type$ + "[" + (.@del - 1) + "]"), 1;
                    mes "Key deleted!";
                    next;
                }
            }
        }
        end;
    }

    function	F_ResetVar	{
        .@type$ = getarg(0);
        .@key_list$[0] = "";
        .@resetname$ = "";

        // Special Reset per Type
        if (.@type$ == "daily") {
            //$Credit_DATE = atoi(gettimestr("%Y%m%d", 10)); // Example: Daily Credit Reset
            .@key_list$[0] = "$daily_reset_keys$";
            .@resetname$ = "Daily";
        }
        else if (.@type$ == "weekly") {
            .@key_list$[0] = "$weekly_reset_keys$";
            .@resetname$ = "Weekly";
        }
        else if (.@type$ == "monthly") {
            .@key_list$[0] = "$monthly_reset_keys$";
            .@resetname$ = "Monthly";
        }
        else {
            debugmes "F_ResetVar: Invalid reset type '" + .@type$ + "'";
            return;
        }

        // Load the actual key list
        deletearray .@reset_keys$[0], getarraysize(.@reset_keys$);
        .@size = getarraysize(getd(.@key_list$[0]));
        for (.@i = 0; .@i < .@size; .@i++) {
            .@reset_keys$[.@i] = getd(.@key_list$[0] + "[" + .@i + "]");
        }

        // Delete from SQL acc_reg_num
        freeloop(1);
        for (.@i = 0; .@i < getarraysize(.@reset_keys$); .@i++) {
            query_sql("DELETE FROM `acc_reg_num` WHERE `key` = '" + escape_sql(.@reset_keys$[.@i]) + "'");
        }

        // Reset variables for all players
        query_sql("SELECT `account_id`, `name` FROM `char`", .@account_id, .@char_name$);
        if (getarraysize(.@account_id) > 0) {
            for (.@j = 0; .@j < getarraysize(.@account_id); .@j++) {
                attachrid .@account_id[.@j];
                if (playerattached()) {
                    for (.@k = 0; .@k < getarraysize(.@reset_keys$); .@k++) {
                        set getd(.@reset_keys$[.@k]), 0;
                        if (.@resetname$ == "Daily") {
                            // TEST
                        }
                        if (.@resetname$ == "Monthly") {
                            #BPEXP = 0;
                            #BPPREMIUM = 0;
                            deletearray #BPClaimedNormal;
                            deletearray #BPClaimedPremium;
                        }
                    }
                }
            }
        }
        freeloop(0);
        announce "[ System ] : " + .@resetname$ + " reset initiated.",bc_all;
        return;
    }
}
