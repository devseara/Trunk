S_CHCTypeCheck:

	set .@opt,getarg(0,0);

		return (countitem(#CHC_MOB) >= #CHC_Qtd ? 1 : 0);
	else
		return (countitem(#CHC_MOB) < #CHC_Qtd ? 1 : 0);

			set #CHC_Qtd,0;

			dispbottom "[Hunting Challenge]: "+(countstr(.@rst$,"No Limit")?"Challenge Failed~":"You still have "+.@rst$+" left to finish the challenge.")+"",0xFFFF00;

S_HCTimer:

	if(getarg(0) < 1) return "^36d900No Limit";
	set .@calc,getarg(0);
	set .@d,(.@calc/86400);
	set .@h,(.@calc%86400)/3600;
	set .@m,(.@calc%3600)/60;
	set .@s,(.@calc%60); 
	set @HCtimer,0;
	
	if(!.@d && !.@h && !.@m && !.@s)
		set .@str$,"^36d900No Limit";
	else
	{
		set .@str$,"^d900a3";
		if(.@d) set .@str$,(.@d?.@d+" day"+(.@d>1?"s":"")+""+(.@h||.@m||.@s?", ":"")+"":"");
		if(.@h) set .@str$,.@str$+""+.@h+"h";
		if(.@m) set .@str$,.@str$+""+(.@h?":":"")+""+(.@m<10&&.@h?"0"+.@m:.@m)+"m";
		if(.@s) set .@str$,.@str$+""+(.@m?":":"")+""+(.@s<10?"0"+.@s:.@s)+"s";
		set @HCtimer,.@calc;
	}
	
	return (!@HCtimer?"^36d900No Limit":.@str$);


OnNPCKillEvent:

		{
			dispbottom "[Hunting Challenge]: The time limit to do the challenge has expired!",0xFFFF00;
			end;
		}
	if(#CHC_Ttd >= #CHC_Qtd){ dispbottom "[Hunting Challenge]: You completed the challenge, come back to claim your reward.",0xFFFF00; end; }
	set #CHC_Ttd,#CHC_Ttd+1;	
	showscript("[Hunting Challenge]: You killed ["+#CHC_Ttd+"/"+#CHC_Qtd+" "+strmobinfo(1,killedrid)+"]");
	dispbottom("[Hunting Challenge]: You killed ["+#CHC_Ttd+"/"+#CHC_Qtd+" "+strmobinfo(1,killedrid)+"]"),0xFFFFFF;	
	if(#CHC_Ttd >= #CHC_Qtd)
	{
		dispbottom "[Hunting Challenge]: You completed the challenge, come back to claim your reward!",0xFFFF00;
		showscript("[Hunting Challenge]: Completed!");
	}
end;

	mes "Point Price: "+(.Hunt_Shop[(.@idx+1)]*@bought_quantity)+"";
	mes "Current Points: "+#CHALLENGEPOINTS+"";
	if(select("- Purchase:- Cancel")==2) close;
	if(#CHALLENGEPOINTS < (.Hunt_Shop[(.@idx+1)]*@bought_quantity)){ clear; mes "[Hunting Challenge]"; mes "You don't have enough challenge points!"; close; }
	if(!checkweight(@bought_nameid,@bought_quantity)){ clear; mes "[Hunting Challenge]"; mes "You are carrying too many items!"; close; }

	close;
