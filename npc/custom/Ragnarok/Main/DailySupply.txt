prontera,113,103,6	script	Daily Supply	10571,{
	callfunc "F_NPC_NAME";
	if(#DailyLogin==gettime(DT_YYYYMMDD)){
		mes "You already received your daily supply reward today.";
		close;
		end;
	}

	F_CheckInventoryCount;

	mes "Here's your daily supply reward points!";
	mes "Keep active and Invite more players. Thank you";
	explode(.@items$,.items$[gettime(DT_DAYOFWEEK)],"|");
	for(.@i=0;.@i<getarraysize(.@items$);.@i+=2)
		getitem atoi(.@items$[.@i]),atoi(.@items$[.@i+1]);
        if (vip_status(VIP_STATUS_ACTIVE)) {
            rentitem 41000,43200;
        }else{
            rentitem 41000,28800;
        }
		set #DailyLogin,gettime(DT_YYYYMMDD);
		#DailyLoginReward = #DailyLoginReward + 1;
		message strcharinfo(0), "You have successfully claimed todays reward.";
end;

OnCheckDay:
	switch(gettime(DT_DAYOFWEEK)){
		case 0: return "Sunday";
		case 1: return "Monday";
		case 2: return "Tuesday";
		case 3: return "Wednesday";
		case 4: return "Thursday";
		case 5: return "Friday";
		case 6: return "Saturday";
		default: return "ERROR!";
			break;
	}
return;

OnInit:
    .Active = 1;
	// Item ID, Amount{,Item ID,Amount}...
	setarray .items$[0],
			"42000|10",	//Sunday
			"42000|5", //Monday
			"42000|5", //Tuesday
			"42000|5", //Wednesday
			"42000|5", //Thursday
			"42000|5", //Friday
			"42000|10"; //Saturday
    while( getvariableofnpc(.Active,"Daily Supply") ) {
    PUBTITLE:
        freeloop(1);
        showscript "Daily Supply [ "+ callsub(OnCheckDay) +" ]",getnpcid(0), AREA;		
        sleep 1000; 
        freeloop(0);
        goto PUBTITLE;
    }
	end;

}
