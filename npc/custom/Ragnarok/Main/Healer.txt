-	script	Healer	-1,{

.@Price = 0;	// Zeny required for heal
.@Buffs = 1;	// Also buff players? (1: yes / 0: no)
.@Delay = 2;	// Heal delay, in seconds
OnTalk:

	if (@HD > gettimetick(2))
		end;
	if (.@Price) {
		message strcharinfo(0), "Healing costs " + callfunc("F_InsertComma",.@Price) + " Zeny.";
		if (Zeny < .@Price)
			end;
		if (select("^0055FFHeal^000000:^777777Cancel^000000") == 2)
			end;
		Zeny -= .@Price;
	}

	emotion 3;
	
	if (.@Delay)
		@HD = gettimetick(2) + .@Delay;

	if (.@Buffs) {
		specialeffect2 EF_HEAL2; percentheal 100,100;
		specialeffect2 EF_INCAGILITY; sc_start SC_INCREASEAGI,900000,10;
		specialeffect2 EF_BLESSING; sc_start SC_BLESSING,900000,10;
		sc_end SC_STONE;
		sc_end SC_FREEZE;
		sc_end SC_STUN;
		sc_end SC_SLEEP;
		sc_end SC_POISON;
		sc_end SC_CURSE;
		sc_end SC_SILENCE;
		sc_end SC_CONFUSION;
		sc_end SC_BLIND;
		sc_end SC_BLOODING;
		sc_end SC_DPOISON;
		sc_end SC_FEAR;
		sc_end SC_COLD;
		sc_end SC_BURNING;
		sc_end SC_DEEP_SLEEP;
		sc_end SC_DEC_AGI;
		sc_end SC_BROKENARMOR;
		sc_end SC_BROKENWEAPON;
		sc_end SC_NOEQUIPWEAPON;
		sc_end SC_NOEQUIPSHIELD;
		sc_end SC_NOEQUIPARMOR;
		sc_end SC_NOEQUIPHELM;
		sc_end SC__STRIPACCESSARY;
		sc_end SC_PROPERTYUNDEAD;
		sc_end SC_ORCISH;
		sc_end SC_BERSERK;
		sc_end SC_SKE;
		sc_end SC_SWOO;
		sc_end SC_SKA;

	if (vip_status(1)) {
		specialeffect2 EF_HEAL2; percentheal 100,100;
		specialeffect2 EF_INCAGILITY; sc_start SC_INCREASEAGI,900000,10;
		specialeffect2 EF_BLESSING; sc_start SC_BLESSING,900000,10;
		specialeffect2 EF_IMPOSITIO; sc_start SC_IMPOSITIO,900000,5;
		specialeffect2 EF_KYRIE; sc_start SC_KYRIE,900000,10;
	
		while(getbrokenid(1))
		    repair(1);

		  set @NonIDCount, 0;
		  getinventorylist;
		  for(set @i, 0; @i < @inventorylist_count; set @i, @i + 1) {
			  if (@inventorylist_identify[@i] == 0) {
				  set @NonIDCount, @NonIDCount + 1;
			  }
		  }
		  if (@NonIDCount >= 1) {
			  for(set @i, 0; @i < @inventorylist_count; set @i, @i + 1) {
			  if (!@inventorylist_identify[@i]) {
				  delitem2 @inventorylist_id[@i],1,0,0,0,0,0,0,0;
				  getitem @inventorylist_id[@i],1;
				  }
			  }
		  }
		}
	}
	end;

OnInit:
.Active = 1;
while( getvariableofnpc(.Active,"Healer") ) {

PUBTITLE:
	freeloop(1);
	showscript "Healer", getnpcid(0), AREA;			
	sleep 1000; 
	freeloop(0);
	goto PUBTITLE;
}
end;

}
