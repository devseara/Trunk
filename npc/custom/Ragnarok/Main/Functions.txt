//=========================================================================
// Equipment Name Display with Cards and Refine
//=========================================================================
function	script	F_GetItemNameFull	{
	.@slt = getitemslots(getarg(2));
	if (getelementofarray(getarg(3), 0) == 255) {
		query_sql("SELECT `name` FROM `char` WHERE `char_id`="+((getelementofarray(getarg(3), 3) << 16) + getelementofarray(getarg(3), 2)), .@forger$);
		switch(getelementofarray(getarg(3), 1) % 1280) {
			case 1:	.@elem$ = "Water";	break;
			case 2:	.@elem$	= "Earth";	break;
			case 3:	.@elem$ = "Fire";	break;
			case 4:	.@elem$ = "Wind";	break;
			default: .@elem$ = "";	break;
		}
		.@vvs$ = "";
		if (getelementofarray(getarg(3), 1) > 1279) {
			for (.@i=0; .@i<(getelementofarray(getarg(3), 1)/1280); .@i++)
				.@vvs$ += "Very ";
			.@vvs$ += "Strong ";
		}
	} else {
		for (.@i=0; .@i<getarraysize(getarg(3)); .@i++) {
			.@found = 0;
			for (.@j=0; .@j < .@i; .@j++) {
				if (getelementofarray(getarg(3), .@i) == getelementofarray(getarg(3), .@j)) 
				.@found++;
			}
			if (.@found == 0) {
				.@cnt = 1;
				for (.@j = .@i + 1; .@j < getarraysize(getarg(3)); .@j++) {
					if (getelementofarray(getarg(3), .@i) == getelementofarray(getarg(3), .@j)) 
					.@cnt++;
				}
				setarray .@cards_cnt[getarraysize(.@cards_cnt)], getelementofarray(getarg(3), .@i), .@cnt;
			}
		}
		for (.@j=0; .@j<getarraysize(.@cards_cnt); .@j+=2) {
			query_sql("SELECT `prefix`, `fix_type` FROM `card_prefix` WHERE `card_id`="+.@cards_cnt[.@j], .@presuf$, .@type);
			if (!.@type) {
				.@card_prefix$ = .@presuf$;
				switch(.@cards_cnt[.@j+1]) {
					case 2:	set .@count$, "Double ";	break;
					case 3:	set .@count$, "Triple ";	break;
					case 4:	set .@count$, "Quadruple ";	break;
					default:	set .@count$, "";	break;
				}
				setarray .@prefix_r$[getarraysize(.@prefix_r$)], .@count$ + .@card_prefix$;
			} else {
				.@card_suffix$ = .@presuf$;
				switch(.@cards_cnt[.@j+1]) {
					case 2:	set .@count_$, " Double";	break;
					case 3:	set .@count_$, " Triple";	break;
					case 4:	set .@count_$, " Quadruple";	break;
					default:	set .@count_$, "";	break;
				}
				setarray .@suffix_r$[getarraysize(.@suffix_r$)], .@card_suffix$ + .@count_$;
			}
		}
		if (getarraysize(.@prefix_r$)) .@prefix$ = implode(.@prefix_r$, " ");
		if (getarraysize(.@suffix_r$)) .@suffix$ = implode(.@suffix_r$, " ");
	}
	.@str$ = "";
	if (getarg(0) > 1) .@str$ += sprintf("%d ", getarg(0));
	if (getarg(1) > 0) .@str$ += sprintf("+%d ", getarg(1));
	if (getarg(4) == 1 && getelementofarray(getarg(3), 0) == 255)
		.@str$ += sprintf("%s %s's ", .@vvs$, .@forger$);
	if (.@elem$ != "") .@str$ += sprintf("%s ", .@elem$);
	if (.@prefix$ != "") .@str$ += sprintf("%s ", .@prefix$);
	.@str$ += sprintf("%s", getitemname(getarg(2)));
	if (.@suffix$ != "") .@str$ += sprintf(" %s", .@suffix$);
	if (.@slt > 0 && getelementofarray(getarg(3), 0) != 255) .@str$ += sprintf(" [%d]", .@slt);
	return .@str$;
}

//=========================================================================
// F_TimeShow(<time>);
//=========================================================================
function	script	F_TimeShow	{
	.@type = getarg(1,0);
	.@sec = getarg(0);
	.@day = .@sec / 86400;
	.@sec -= (.@day * 86400);
	.@hr = .@sec / 3600;
	.@sec -= (.@hr * 3600);
	.@min = .@sec / 60;
	.@sec -= (.@min * 60);

	.@str$ = "";

	.@day$ = .@type ? sprintf("%dd", .@day) : F_InsertPlural(.@day, "day", 0);
	.@hr$  = .@type ? sprintf("%dh", .@hr) : F_InsertPlural(.@hr, "hour", 0);
	.@min$ = .@type ? sprintf("%dmn", .@min) : F_InsertPlural(.@min, "minute", 0);
	.@sec$ = .@type ? sprintf("%ds", .@sec) : F_InsertPlural(.@sec, "second", 0);

	.@str$ += (.@day > 0) ? .@day$ + "" : "";
	.@str$ += (.@day > 0 && (.@hr > 0 || .@min > 0 || .@sec > 0)) ? ", " : "";
	.@str$ += (.@hr > 0) ? .@hr$ + "" : "";
	.@str$ += (.@hr > 0 && (.@min > 0 || .@sec > 0)) ? ", " : "";
	.@str$ += (.@min > 0) ? .@min$ + "" : "";
	.@str$ += (.@min > 0 && .@sec > 0) ? ", " : "";
	.@str$ += (.@sec > 0) ? .@sec$ : "";

	if (charat(.@str$, 0) == " ")
		delchar(.@str$, 0);

	if (charat(.@str$, getstrlen(.@str$) - 1) == " ")
		delchar(.@str$, 0);

	return .@str$;
}

//=========================================================================
// F_TimeDisp(<time>);
//=========================================================================
function	script	F_TimeDisp	{
	.@sec =  getarg(0) - gettimetick(2);
	.@day = .@sec / 86400;
	.@sec = .@sec - (.@day * 60 * 60 * 24);
	.@hr  = .@sec / 3600;
	.@sec = .@sec - (.@hr * 60 * 60);
	.@min = .@sec / 60;
	.@sec = .@sec - (.@min * 60);

	.@str$ = "";

	.@day$ = F_InsertPlural(.@day, "day", 0, "%d %s");
	.@hr$  = F_InsertPlural(.@hr, "hour", 0, "%d %s");
	.@min$ = F_InsertPlural(.@min, "minute", 0, "%d %s");

	.@str$ = (.@day > 0 ? .@day$+" ":"")+""+(.@hr > 0 ? .@hr$+" ":"")+""+(.@min > 0 ? .@min$+" ":"");
	return .@str$;
}

// Dynamic Sending RODEX Mail
// Syntax: 
// F_RODEX_SendMail(<type>, <char_id>, <sender>, <title>, <message>, <zeny>{, <itm>, <amt>, <ref>, <crd0>, <crd1>, <crd2>, <crd3>});

function	script	F_RODEX_SendMail	{
	.@type      = getarg(0);
	.@cid       = getarg(1);
	.@sender$   = getarg(2, "Kafrashop.net");
	.@title$    = getarg(3, "System Announcement");
	.@msg$      = getarg(4, "Hello World!\r\n\r\n- Kafrashop.net");
	.@zeny      = getarg(5);
	.@itm       = getarg(6, 0);
	.@qty       = getarg(7, 0);
	.@ref       = getarg(8, 0);
	for (.@i=0; .@i<4; .@i++)
		setd(".@crd"+.@i, getarg(9+.@i, 0));
	if (.@itm && .@qty) {
		if (.@type == 1) {
			freeloop(1);
			while (.@qty > 0) {
				setarray .@itm_r[0], .@itm;
				setarray .@amt_r[0], .@qty > 30000 ? 30000 : .@qty;
				mail(.@cid, .@sender$, .@title$, .@msg$, 0, .@itm_r, .@amt_r);
				.@qty -= 30000;
			}
			freeloop(0);
		}
		else if (.@type == 2){
			freeloop(1);
			do {
				setarray .@itm_r[0], .@itm, .@itm, .@itm, .@itm, .@itm;
				setarray .@amt_r[0], 1, 1, 1, 1, 1;
				setarray .@ref_r[0], .@ref, .@ref, .@ref, .@ref, .@ref;
				setarray .@bnd_r[0], 0, 0, 0, 0, 0;
				setarray .@crd0_r[0], .@crd0, .@crd0, .@crd0, .@crd0, .@crd0;
				setarray .@crd1_r[0], .@crd1, .@crd1, .@crd1, .@crd1, .@crd1;
				setarray .@crd2_r[0], .@crd2, .@crd2, .@crd2, .@crd2, .@crd2;
				setarray .@crd3_r[0], .@crd3, .@crd3, .@crd3, .@crd3, .@crd3;
				switch (.@qty) {
					default:
						mail(.@cid, .@sender$, .@title$, .@msg$, 0, .@itm_r, .@amt_r, .@ref_r, .@bnd_r, .@crd0_r, .@crd1_r, .@crd2_r, .@crd3_r);
						break;
					case 1:
						deletearray(.@itm_r[0], 4);
						deletearray(.@amt_r[0], 4);
						deletearray(.@ref_r[0], 4);
						deletearray(.@bnd_r[0], 4);
						deletearray(.@crd0_r[0], 4);
						deletearray(.@crd1_r[0], 4);
						deletearray(.@crd2_r[0], 4);
						deletearray(.@crd3_r[0], 4);
						mail(.@cid, .@sender$, .@title$, .@msg$, 0, .@itm_r, .@amt_r, .@ref_r, .@bnd_r, .@crd0_r, .@crd1_r, .@crd2_r, .@crd3_r);
						break;
					case 2:
						deletearray(.@itm_r[0], 3);
						deletearray(.@amt_r[0], 3);
						deletearray(.@ref_r[0], 3);
						deletearray(.@bnd_r[0], 3);
						deletearray(.@crd0_r[0], 3);
						deletearray(.@crd1_r[0], 3);
						deletearray(.@crd2_r[0], 3);
						deletearray(.@crd3_r[0], 3);
						mail(.@cid, .@sender$, .@title$, .@msg$, 0, .@itm_r, .@amt_r, .@ref_r, .@bnd_r, .@crd0_r, .@crd1_r, .@crd2_r, .@crd3_r);
						break;
					case 3:
						deletearray(.@itm_r[0], 2);
						deletearray(.@amt_r[0], 2);
						deletearray(.@ref_r[0], 2);
						deletearray(.@bnd_r[0], 2);
						deletearray(.@crd0_r[0], 2);
						deletearray(.@crd1_r[0], 2);
						deletearray(.@crd2_r[0], 2);
						deletearray(.@crd3_r[0], 2);
						mail(.@cid, .@sender$, .@title$, .@msg$, 0, .@itm_r, .@amt_r, .@ref_r, .@bnd_r, .@crd0_r, .@crd1_r, .@crd2_r, .@crd3_r);
						break;
					case 4:
						deletearray(.@itm_r[0], 1);
						deletearray(.@amt_r[0], 1);
						deletearray(.@ref_r[0], 1);
						deletearray(.@bnd_r[0], 1);
						deletearray(.@crd0_r[0], 1);
						deletearray(.@crd1_r[0], 1);
						deletearray(.@crd2_r[0], 1);
						deletearray(.@crd3_r[0], 1);
						mail(.@cid, .@sender$, .@title$, .@msg$, 0, .@itm_r, .@amt_r, .@ref_r, .@bnd_r, .@crd0_r, .@crd1_r, .@crd2_r, .@crd3_r);
						break;
				}
				.@qty -= 5;
			} while(.@qty > 0);
			freeloop(0);
		} else
			return;
	} else {
		freeloop(1);
		while (.@zeny > 0) {
			mail(.@cid, .@sender$, .@title$, .@msg$, .@zeny > MAX_ZENY ? MAX_ZENY : .@zeny);
			.@zeny -= MAX_ZENY;
		}
		freeloop(0);
	}
	return;
}

//=========================================================================
// F_GetRate(<amt>, <percent>);
//=========================================================================
function	script	F_GetRate	{
	.@amt = getarg(0);
	.@rate = getarg(1);
	if (.@amt > 999)
		return (.@amt / 1000 * .@rate) * 10;
	if (.@amt < 1000 && .@amt > 99)
		return (.@amt / 100) * .@rate;
	return (.@amt / 10) * (.@rate / 10);
}

function	script	F_PointGain	{
	.@type = getarg(0, 0);
	.@pt = getarg(1, 0);
	.@evt$ = getarg(2);
	if (.@pt) {
		if (!.@type)
			.@str$ = sprintf("You won %s from %s.", F_CommaPlural(.@pt, "Event Point"), .@evt$);
		else
			.@str$ = sprintf("You received %s as a consolation from %s.", F_CommaPlural(.@pt, "Event Point"), .@evt$);
		message(strcharinfo(0), .@str$);
	}
	return;
}
